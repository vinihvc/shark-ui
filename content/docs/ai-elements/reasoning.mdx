## Usage

```tsx
import {
  Reasoning,
  ReasoningContent,
  ReasoningTrigger,
} from "@/components/ui/reasoning";
```

```tsx
<Reasoning duration={8}>
  <ReasoningTrigger duration={8} />
  <ReasoningContent>The helper already checks empty fields...</ReasoningContent>
</Reasoning>
```

Place reasoning above the assistant [Bubble](/docs/ai-elements/bubble) inside [Message](/docs/ai-elements/message).
