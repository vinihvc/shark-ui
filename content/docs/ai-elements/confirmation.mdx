## Usage

```tsx
import {
  Confirmation,
  ConfirmationAction,
  ConfirmationActions,
  ConfirmationRequest,
  ConfirmationTitle,
} from "@/components/ui/confirmation";
```

```tsx
<Confirmation state="request">
  <ConfirmationRequest>
    <ConfirmationTitle>Allow write to helpers.ts?</ConfirmationTitle>
  </ConfirmationRequest>
  <ConfirmationActions>
    <ConfirmationAction variant="outline">Reject</ConfirmationAction>
    <ConfirmationAction>Approve</ConfirmationAction>
  </ConfirmationActions>
</Confirmation>
```

Use this in the transcript, not a modal. Pair with [Tool](/docs/ai-elements/tool).
