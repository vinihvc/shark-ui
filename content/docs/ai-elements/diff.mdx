## Usage

```tsx
import { Diff, DiffContent, DiffHeader, DiffLine } from "@/components/ui/diff";
```

```tsx
<Diff>
  <DiffHeader>src/utils/helpers.ts</DiffHeader>
  <DiffContent>
    <DiffLine type="delete">  return Boolean(email);</DiffLine>
    <DiffLine type="add">  return isValidEmail(email);</DiffLine>
  </DiffContent>
</Diff>
```

Use in a [Tool](/docs/ai-elements/tool) result or beside the transcript as an artifact.
