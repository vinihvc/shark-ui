## Anatomy

```text
Diff
├── DiffHeader
│   ├── DiffFile
│   └── DiffStats
└── DiffContent
    └── DiffLine
```

## Usage

```tsx
import {
  Diff,
  DiffContent,
  DiffFile,
  DiffHeader,
  DiffLine,
  DiffStats,
} from "@/components/ui/diff";
```

```tsx
<Diff>
  <DiffHeader>
    <DiffFile>src/auth.ts</DiffFile>
    <DiffStats added={4} removed={1} />
  </DiffHeader>
  <DiffContent>
    <DiffLine line={13} type="delete">  return localStorage.token;</DiffLine>
    <DiffLine line={13} type="add">  const t = cookies.get("session");</DiffLine>
  </DiffContent>
</Diff>
```

## API Reference

### Diff

Root panel for a unified diff.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### DiffHeader

Header for the file path and stats.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### DiffFile

File path. Children are the path text.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### DiffStats

Added and removed counts.

| Prop | Type | Default |
| --- | --- | --- |
| `added` | `number` | `0` |
| `removed` | `number` | `0` |
| `aria-label` | `string` | `"{added} added, {removed} removed"` |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### DiffContent

Scrollable hunk. Line numbers and `+`/`-` markers sit in a sticky gutter on horizontal overflow.

| Prop | Type | Default |
| --- | --- | --- |
| `dir` | `"ltr" \| "rtl"` | `"ltr"` |
| `className` | `string` | `-` |

### DiffLine

A source line. Children are the line text. The `+`/`-` marker comes from `type`.

| Prop | Type | Default |
| --- | --- | --- |
| `type` | `"add" \| "delete" \| "context"` | `"context"` |
| `line` | `number \| null` | `-` |
| `aria-label` | `string` | From `type` and `line` |
| `className` | `string` | `-` |
