## Usage

```tsx
import { Task, TaskContent, TaskItemFile, TaskTrigger } from "@/components/ui/task";
```

```tsx
<Task status="in-progress">
  <TaskTrigger status="in-progress" title="Patch isValidEmail" />
  <TaskContent>
    <TaskItemFile>src/utils/helpers.ts</TaskItemFile>
  </TaskContent>
</Task>
```
