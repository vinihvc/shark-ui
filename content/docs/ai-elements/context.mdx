## Usage

```tsx
import {
  Context,
  ContextBody,
  ContextContent,
  ContextFooter,
  ContextHeader,
  ContextInputUsage,
  ContextOutputUsage,
  ContextTrigger,
} from "@/components/ui/context";
```

```tsx
<Context maxTokens={128000} usedTokens={18420} costLabel="$0.042">
  <ContextTrigger />
  <ContextContent>
    <ContextHeader />
    <ContextBody>
      <ContextInputUsage />
      <ContextOutputUsage />
    </ContextBody>
    <ContextFooter />
  </ContextContent>
</Context>
```

Pass numeric `usedTokens` and `maxTokens`. Optional `usage` breaks down input, output, reasoning, and cache. Optional `costLabel` is display-only.
